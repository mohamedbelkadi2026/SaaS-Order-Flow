━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
In server/services/city-aliases.ts, find matchCityId function.

The problem: order city = "OUJDA", vitips_cities table has nameNorm = "oujda" and externalId = "Oujda".
matchCityId should find it via nameNorm matching but may be failing.

Also in server/storage.ts, getVitipsCityAbbr(), the matching uses matchCityId which returns externalId.
The Vitipsexpress API requires city = abbr (e.g. "Oujda").

Add a fallback in getVitipsCityAbbr: if matchCityId returns null, try direct case-insensitive match:

async getVitipsCityAbbr(storeId: number, cityName: string): Promise<string | null> {
  if (!(cityName || "").trim()) return null;
  const cities = await db.select().from(vitipsCities).where(eq(vitipsCities.storeId, storeId));
  
  // Try matchCityId first
  const matched = matchCityId(cities, cityName);
  if (matched) return matched;
  
  // Fallback: case-insensitive name match
  const norm = cityName.toLowerCase().trim();
  const found = cities.find(c => 
    c.nameNorm === norm ||
    c.name.toLowerCase() === norm ||
    c.externalId.toLowerCase() === norm
  );
  return found?.externalId ?? null;
}
