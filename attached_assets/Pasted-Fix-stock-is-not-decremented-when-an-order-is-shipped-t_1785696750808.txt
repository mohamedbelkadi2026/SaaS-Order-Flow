PROBLEM:
When clicking "Expédier" and the order is successfully shipped (status → "Attente De Ramassage"),
the product stock in the inventory is never reduced. Stock only changes via manual edits.

SOLUTION — Add stock decrement after successful shipment:

Add this function to the DatabaseStorage class:

async decrementStockForOrder(orderId: number, storeId: number): Promise<void> {
  // Get order items
  const items = await db.select().from(orderItems).where(eq(orderItems.orderId, orderId));
  if (!items.length) return;

  for (const item of items) {
    if (!item.productId) continue;
    const qty = item.quantity || 1;

    // Try to match variant first (by SKU or variant name)
    const variants = await db.select().from(productVariants)
      .where(eq(productVariants.productId, item.productId));

    let variantMatched = false;
    if (variants.length > 0 && item.variantInfo) {
      // Try to find variant by name match
      const variantName = (item.variantInfo || '').trim();
      const matched = variants.find(v =>
        v.name === variantName ||
        v.sku === item.sku ||
        v.name?.includes(variantName) ||
        variantName.includes(v.name || '')
      );
      if (matched) {
        // Decrement variant stock (don't go below 0)
        const newStock = Math.max(0, (matched.stock || 0) - qty);
        await db.update(productVariants)
          .set({ stock: newStock })
          .where(eq(productVariants.id, matched.id));
        // Log stock movement
        await db.insert(stockMovements).values({
          storeId,
          productId: item.productId,
          type: 'shipped',
          quantity: -qty,
          reason: `Expédition commande #${orderId} (variant: ${matched.name})`,
          orderId,
        });
        variantMatched = true;
      }
    }

    if (!variantMatched) {
      // Decrement parent product stock
      const [product] = await db.select().from(products).where(eq(products.id, item.productId));
      if (!product) continue;
      const newStock = Math.max(0, (product.stock || 0) - qty);
      await db.update(products)
        .set({ stock: newStock })
        .where(eq(products.id, item.productId));
      // Log stock movement
      await db.insert(stockMovements).values({
        storeId,
        productId: item.productId,
        type: 'shipped',
        quantity: -qty,
        reason: `Expédition commande #${orderId}`,
        orderId,
      });
    }
  }
}
