server/services/profit.ts — remplacer le bloc `if (isDel) { ... }` de la
boucle "Bucket orders that have NO orderItems rows" par :

```ts
if (isDel) {
  s.revenue      += Number((order as any).totalPrice   || 0) / 100;
  s.shippingCost += Number((order as any).shippingCost || 0) / 100;

  // ── NOUVEAU : calculer le vrai coût en splittant "A + B + C" et en
  // faisant correspondre CHAQUE partie à un produit/variante du catalogue,
  // avec la même cascade que pour les commandes déjà liées (exact → base+
  // suffixe variante → fuzzy contains → coût de variante en dernier
  // recours). Remplace le fallback order.productCost (toujours à 0 ici).
  let computedCost = 0;
  if (raw) {
    const parts = raw.split(/\s*\+\s*/).map((p: string) => p.replace(/\s*\(non li[ée]\)\s*$/i, '').trim()).filter(Boolean);
    for (const part of parts) {
      let partCost = costByName.get(norm(part)) || 0;

      if (!partCost) {
        const { base: partBase, suffix: partSuffix } = splitVariant(part);
        const suffixGuess = partSuffix || part.trim().split(' ').pop();
        const baseGuess = partSuffix ? partBase : part.replace(new RegExp(`\\s+${suffixGuess}$`), '').trim();
        if (baseGuess && norm(baseGuess) !== norm(part)) {
          const altProductId = idByName.get(norm(baseGuess));
          if (altProductId) {
            const altVariantCost = suffixGuess ? variantCostByKey.get(`${altProductId}::${norm(suffixGuess)}`) : undefined;
            partCost = altVariantCost || costByName.get(norm(baseGuess)) || 0;
          }
        }
        // Fuzzy contains match, en dernier recours — même logique que plus haut.
        if (!partCost) {
          const partNorm = norm(part);
          const baseNorm = norm(baseGuess || part);
          const fuzzy = storeProductsList
            .filter(p => {
              const pn = norm(p.name);
              return pn.length >= 4 && (partNorm.includes(pn) || pn.includes(baseNorm) || baseNorm.includes(pn));
            })
            .sort((a, b) => b.name.length - a.name.length)[0];
          if (fuzzy) {
            const altVariantCost = suffixGuess ? variantCostByKey.get(`${fuzzy.id}::${norm(suffixGuess)}`) : undefined;
            partCost = altVariantCost || Number(fuzzy.costPrice) || 0;
          }
        }
      }
      computedCost += Number(partCost) || 0;
    }
  }
  // Si rien n'a matché (vrai "— Sans produit —" sans nom du tout, ou aucune
  // partie reconnue), garder le fallback existant order.productCost.
  s.productCost += computedCost > 0 ? (computedCost / 100) : (Number((order as any).productCost || 0) / 100);

  const agentDH   = agentRateMap.get((order as any).assignedToId) ?? 0;
  s.confirmationCost += agentDH;
}
```

Ce fix réutilise `costByName`, `idByName`, `variantCostByKey`,
`storeProductsList` et `splitVariant` déjà chargés/importés plus haut dans
le même fichier — aucune nouvelle requête DB, aucun nouvel import
nécessaire.
