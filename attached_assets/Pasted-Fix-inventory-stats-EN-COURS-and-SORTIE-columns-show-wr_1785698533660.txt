PROBLEM 1: inTransitItems query only counts these statuses:
['in_progress', 'expédié', 'Attente De Ramassage']

This misses: 'transit', 'unreachable', 'En Cours De Retour', 'refused', 'Retour Recu'

Fix — replace the inTransitItems query statuses with ALL active shipping statuses:

const EN_COURS_STATUSES = [
  'in_progress',
  'expédié', 
  'Attente De Ramassage',
  'transit',
  'unreachable',
  'En Cours De Retour',
  'refused',
  'Retour Recu',
];

const inTransitItems = await db.select({ qty: sql<number>`COALESCE(SUM(${orderItems.quantity}), 0)` })
  .from(orderItems)
  .innerJoin(orders, eq(orderItems.orderId, orders.id))
  .where(and(
    eq(orderItems.productId, p.id),
    eq(orders.storeId, storeId),
    inArray(orders.status, EN_COURS_STATUSES)
  ));
